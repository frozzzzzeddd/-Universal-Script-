local function create(class, props, children)
	local inst = Instance.new(class)
	for k, v in pairs(props or {}) do
		inst[k] = v
	end
	for _, child in ipairs(children or {}) do
		child.Parent = inst
	end
	return inst
end

local function tween(obj, time, props, style, dir)
	local t = TweenService:Create(
		obj,
		TweenInfo.new(time, style or Enum.EasingStyle.Quad, dir or Enum.EasingDirection.Out),
		props
	)
	t:Play()
	return t
end

local function corner(r)
	return create("UICorner", { CornerRadius = UDim.new(0, r) })
end

local function stroke(color, thickness, transparency)
	return create("UIStroke", {
		Color = color,
		Thickness = thickness or 1,
		Transparency = transparency or 0,
		ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
	})
end

local function gradient(c1, c2, rotation)
	return create("UIGradient", {
		Color = ColorSequence.new(c1, c2),
		Rotation = rotation or 0,
	})
end

local function getParent()
	local ok, ui = pcall(function()
		return (gethui and gethui()) or CoreGui
	end)
	if ok and ui then return ui end
	return Players.LocalPlayer:WaitForChild("PlayerGui")
end

local function readSavedKey()
	if CONFIG.SaveKey and isfile and readfile and isfile(CONFIG.SaveFile) then
		local ok, data = pcall(readfile, CONFIG.SaveFile)
		if ok then return data end
	end
end

local function saveKey(key)
	if CONFIG.SaveKey and writefile then
		pcall(writefile, CONFIG.SaveFile, key)
	end
end

local function isValid(key)
	for _, k in ipairs(CONFIG.ValidKeys) do
		if key == k then return true end
	end
	return false
end

-- Button factory
local function makeButton(name, text, xScale, xOffset, filled)
	local btn = create("TextButton", {
		Name = name,
		Position = UDim2.new(xScale, xOffset, 0, 168),
		Size = UDim2.new(0.5, -28, 0, 44),
		BackgroundColor3 = filled and Color3.new(1, 1, 1) or CONFIG.Panel,
		Text = "",
		AutoButtonColor = false,
		Parent = main,
	}, { corner(10) })

	if filled then
		gradient(CONFIG.Accent, CONFIG.Accent2, 25).Parent = btn
	else
		stroke(CONFIG.Accent, 1.5, 0.2).Parent = btn
	end

	local label = create("TextLabel", {
		Size = UDim2.new(1, 0, 1, 0),
		BackgroundTransparency = 1,
		Text = text,
		Font = Enum.Font.GothamBold,
		TextSize = 14,
		TextColor3 = Color3.new(1, 1, 1),
		Parent = btn,
	})

	btn.MouseEnter:Connect(function()
		tween(btn, 0.15, { Size = UDim2.new(0.5, -24, 0, 46) })
		if not filled then
			tween(btn, 0.15, { BackgroundColor3 = Color3.fromRGB(36, 32, 54) })
		end
	end)
	btn.MouseLeave:Connect(function()
		tween(btn, 0.15, { Size = UDim2.new(0.5, -28, 0, 44) })
		if not filled then
			tween(btn, 0.15, { BackgroundColor3 = CONFIG.Panel })
		end
	end)

	return btn, label
end

local function setStatus(text, color)
	status.Text = text
	tween(status, 0.2, { TextColor3 = color or CONFIG.SubText })
end

local function shake(obj)
	local base = obj.Position
	for i = 1, 3 do
		tween(obj, 0.04, { Position = base + UDim2.new(0, 8, 0, 0) })
		task.wait(0.04)
		tween(obj, 0.04, { Position = base - UDim2.new(0, 8, 0, 0) })
		task.wait(0.04)
	end
	tween(obj, 0.04, { Position = base })
end

local function closeGui()
	tween(main, 0.25, { Size = UDim2.new(0, 440, 0, 0) }, Enum.EasingStyle.Back, Enum.EasingDirection.In)
	tween(glow, 0.25, { ImageTransparency = 1 })
	task.wait(0.3)
	gui:Destroy()
end

local busy = false

checkBtn.MouseButton1Click:Connect(function()
	if busy then return end
	busy = true

	local key = keyBox.Text:gsub("^%s+", ""):gsub("%s+$", "")
	checkLabel.Text = "Checking..."
	setStatus("Verifying your key...", CONFIG.SubText)
	task.wait(0.6)

	if isValid(key) then
		saveKey(key)
		checkLabel.Text = "✔  Success"
		tween(inputStroke, 0.2, { Color = CONFIG.Success })
		setStatus("Key accepted! Loading Jua...", CONFIG.Success)
		task.wait(0.9)
		closeGui()
		task.spawn(OnSuccess)
	else
		checkLabel.Text = "✔  Check Key"
		tween(inputStroke, 0.2, { Color = CONFIG.Error })
		setStatus(key == "" and "Please enter a key first." or "Invalid key. Try again.", CONFIG.Error)
		shake(inputFrame)
		task.wait(0.8)
		tween(inputStroke, 0.3, { Color = Color3.fromRGB(50, 50, 70) })
		busy = false
	end
end)

copyBtn.MouseButton1Click:Connect(function()
	local copier = setclipboard or toclipboard or (Clipboard and Clipboard.set)
	if copier then
		pcall(copier, CONFIG.KeyLink)
		copyLabel.Text = "✔  Copied!"
		setStatus("Link copied to clipboard.", CONFIG.Success)
	else
		setStatus("Clipboard unsupported. Link: " .. CONFIG.KeyLink, CONFIG.Error)
	end
	task.delay(1.5, function()
		if copyLabel and copyLabel.Parent then
			copyLabel.Text = "⧉  Copy Link"
		end
	end)
end)

-- Dragging
do
	local dragging, dragStart, startPos
	header.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			dragStart = input.Position
			startPos = main.Position
		end
	end)
	UserInputService.InputChanged:Connect(function(input)
		if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
			or input.UserInputType == Enum.UserInputType.Touch) then
			local delta = input.Position - dragStart
			local pos = UDim2.new(
				startPos.X.Scale, startPos.X.Offset + delta.X,
				startPos.Y.Scale, startPos.Y.Offset + delta.Y
			)
			main.Position = pos
			glow.Position = pos
		end
	end)
	UserInputService.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
			dragging = false
		end
	end)
end
